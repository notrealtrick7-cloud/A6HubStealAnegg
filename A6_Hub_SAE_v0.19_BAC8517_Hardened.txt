local function protect(obj)

local BAD_REMOTE_KEYS = {
    "kick","ban","detect","flag","report","anticheat","ac_","threat",
    "telemetry","log","punish","cheat","exploit","speed","velocity",
    "position","teleport","tp","moderation","mod_","security","secure",

}

local function isBadRemote(name)
    name = tostring(name or ""):lower()
    for _, k in ipairs(BAD_REMOTE_KEYS) do
        if name:find(k, 1, true) then return true end
    end
    return false
end

if mt and setreadonly then
    setreadonly(mt, false)
    oldNamecall = mt.__namecall
    oldIndex    = mt.__index
    oldNewIndex = mt.__newindex

    mt.__namecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()
        if typeof(self) == "Instance" then
            local n = self.Name
            if method == "FireServer" or method == "InvokeServer" or method == "Fire" or method == "Invoke" then
                if isBadRemote(n) then return end
                local parentName = self.Parent and self.Parent.Name or ""
                if isBadRemote(parentName) then return end
            end
        end
        return oldNamecall(self, ...)
    end)

    mt.__index = newcclosure(function(self, k)
        if protected[self] then
            if k == "WalkSpeed" then return 16 end
            if k == "JumpPower" then return 50 end
            if k == "JumpHeight" then return 7.2 end
            if k == "HipHeight" then return 0 end   -- AC always sees 0
        end
        return oldIndex(self, k)
    end)

    mt.__newindex = newcclosure(function(self, k, v)
        if protected[self] then
            if k == "WalkSpeed" or k == "JumpPower" or k == "JumpHeight" then
                -- allow real write, fake on read
                return oldNewIndex(self, k, v)
            elseif k == "HipHeight" then
                -- allow real write for air-walk, fake on read always 0
                return oldNewIndex(self, k, v)
            end
        end
        return oldNewIndex(self, k, v)
    end)
    setreadonly(mt, true)
end

local function jitter(base, lo, hi)
    lo = lo or -0.12
    hi = hi or 0.12
    return base + lo + math.random() * (hi - lo)
end

local function acWait(base)
    task.wait(jitter(base, base * 0.25, base * 1.20))
end

local function acWalkSpeed(hum, target)
    if not hum then return end
    protect(hum)
    local real = math.clamp(target + math.random(-1, 1), 14, 22)
    pcall(function() hum.WalkSpeed = real end)
end

local function setHipHeight(val)
    local hum = humanoid()

        hum.HipHeight = val

end

local function raiseHip()

end

local function lowerHip()

    hipRaised = false
end

local function stopMove()
    local hum = humanoid()
    if not hum then return end
    pcall(function() hum:MoveTo(hrp() and hrp().Position or Vector3.zero) end)
end

local function walkTo(goal, timeout)
    local root = hrp()
    local hum = humanoid()
    if not root or not hum then return false end
    timeout = timeout or 16
    local gp = typeof(goal) == "CFrame" and goal.Position or goal
    gp = Vector3.new(gp.X, root.Position.Y, gp.Z)
    acWalkSpeed(hum, S.StealWalkSpeed or 16)
    pcall(function() hum:MoveTo(gp) end)
    local t0 = tick()
    while tick() - t0 < timeout do
        root = hrp()
        if not root then return false end
        local d = (Vector3.new(root.Position.X, 0, root.Position.Z)
                 - Vector3.new(gp.X, 0, gp.Z)).Magnitude
        if d < 4 then stopMove(); return true end
        if tick() - t0 > 2.5 and d > 12 then
            pcall(function()
                hum:MoveTo(gp + Vector3.new(math.random(-2, 2), 0, math.random(-2, 2)))
            end)
        end
        task.wait(jitter(0.14, 0.05, 0.09))
    end
    stopMove()
    return false
end

local function softTweenTo(cf, speed)
    local root = hrp()
    if not root then return end
    speed = math.clamp(speed or S.TweenSpeed or 32, 20, 48)
    local tp = typeof(cf) == "CFrame" and cf.Position or cf
    tp = Vector3.new(tp.X, root.Position.Y + jitter(0.6, -0.15, 0.25), tp.Z)
    local dist = (root.Position - tp).Magnitude
    if dist < 2.5 then return end
    if dist > 55 then
        local steps = math.ceil(dist / 38)
        local dir = (tp - root.Position)
        for i = 1, steps do
            local mid = root.Position + dir * (i / steps)
            softTweenTo(CFrame.new(mid), speed)
            acWait(0.07)
            root = hrp()
            if not root then return end
        end
        return
    end
    local t = math.clamp(dist / speed, 0.22, 6.0)
    local tw = TweenService:Create(root, TweenInfo.new(t, Enum.EasingStyle.Sine), {CFrame = CFrame.new(tp)})
    tw:Play()
    local done, t0 = false, tick()
    tw.Completed:Connect(function() done = true end)
    while not done and tick() - t0 < t + 0.55 do task.wait(0.04) end
end

local function tweenTo(cf, speed)
    if S.MoveStyle == "Walk" then
        walkTo(typeof(cf) == "CFrame" and cf.Position or cf, 16)
    else
        softTweenTo(cf, speed)
    end
end

-- short HipHeight air segment only
local function hipTweenTo(cf, speed)
    local root = hrp()
    if not root then return end
    raiseHip()
    speed = math.clamp(speed or (S.TweenSpeed or 32) + 6, 24, 50)
    local tp = typeof(cf) == "CFrame" and cf.Position or cf
    local elevY = root.Position.Y
    tp = Vector3.new(tp.X, elevY, tp.Z)
    local dist = (Vector3.new(root.Position.X, 0, root.Position.Z) - Vector3.new(tp.X, 0, tp.Z)).Magnitude
    if dist < 3 then lowerHip(); return end
    if dist > 60 then
        local steps = math.ceil(dist / 40)
        local dir = (tp - root.Position)
        for i = 1, steps do
            local mid = root.Position + dir * (i / steps)
            mid = Vector3.new(mid.X, elevY, mid.Z)
            softTweenTo(CFrame.new(mid), speed)
            acWait(0.06)
            root = hrp()
            if not root then lowerHip(); return end
            elevY = root.Position.Y
        end
        lowerHip()
        return
    end
    local t = math.clamp(dist / speed, 0.18, 4.8)
    local tw = TweenService:Create(root, TweenInfo.new(t, Enum.EasingStyle.Sine), {CFrame = CFrame.new(tp)})
    tw:Play()
    local done, t0 = false, tick()
    tw.Completed:Connect(function() done = true end)
    while not done and tick() - t0 < t + 0.4 do task.wait(0.035) end
    lowerHip()
end

end

            if p then return p end
        end
    end
end

    local order = {"divine","eternal","secret","cosmic","mythic","legendary","epic","rare","uncommon","common"}
    local map = {divine="Divine",eternal="Eternal",secret="Secret",cosmic="Cosmic",mythic="Mythic",
                 legendary="Legendary",epic="Epic",rare="Rare",uncommon="Uncommon",common="Common"}
    local result = "Common"
    for _, k in ipairs(order) do if text:find(k, 1, true) then result = map[k]; break end end
    c = c or {}; c.t = now; c.r = result; PetDetectCache[inst] = c
    return result
end

local function detectPetName(inst)
    if not inst then return nil end
    local now = tick()
    local c = PetDetectCache[inst]
    if c and now - c.t < PET_CACHE_TTL and c.pet ~= nil then return c.pet or nil end
    local text = tostring(inst.Name or "")
    local a, b = inst:GetAttribute("PetName"), inst:GetAttribute("Name")
    if a then text = text .. " " .. tostring(a) end
    if b then text = text .. " " .. tostring(b) end
    for _, d in ipairs(inst:GetChildren()) do
        if d:IsA("StringValue") or d:IsA("TextLabel") then
            text = text .. " " .. tostring(d.Text or d.Value or "")
        end
    end
    text = text:lower()
    local best, bestLen = nil, 0
    for _, pet in ipairs(S.PetCatalog) do
        local key = pet.n:lower()
        if #key > bestLen and text:find(key, 1, true) then best, bestLen = pet.n, #key end
    end
    c = c or {}; c.t = now; c.pet = best or false; PetDetectCache[inst] = c
    return best
end

local function isAllowed(rarity, petName)
    rarity = rarity or "Common"
    if S.BlacklistRarity[rarity] then return false end
    if petName and S.PetBlacklist[petName:lower()] then return false end
    if petName and S.PetWhitelist[petName:lower()] then return true end
    return S.WhitelistRarity[rarity] == true
end

local EggPromptCache = {t = 0, list = {}}
local _eggNextScan = 0

local function _anyFeatureActive()
    return S.AutoSteal or S.AutoHatch or S.AutoPlace or S.AutoIndex
        or S.AutoScramble or S.EggESP or S.AutoButterfly
end

local function collectEggPrompts(force)
    local now = tick()
    if not force and not _anyFeatureActive() then return EggPromptCache.list end
    if not force and now < _eggNextScan then return EggPromptCache.list end
    local list = {}
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("ProximityPrompt") then
            local n = (v.Name .. " " .. (v.Parent and v.Parent.Name or "")):lower()
            if n:find("egg", 1, true) or n:find("steal", 1, true)
               or n:find("grab", 1, true) or n:find("pick", 1, true) or n:find("take", 1, true) then
                local obj = v.Parent or v
                table.insert(list, {prompt = v, rarity = detectRarity(obj), parent = v.Parent, pet = detectPetName(obj)})
            end
        end
    end
    for i = #list, 2, -1 do
        local j = math.random(i)
        list[i], list[j] = list[j], list[i]
    end
    _eggNextScan = now + 6.5 + math.random() * 3.0
    EggPromptCache.t = now
    EggPromptCache.list = list
    return list
end

local function nearestAllowedEgg()
    local root = hrp()
    if not root then return nil end
    local best, dist = nil, math.huge
    for _, e in ipairs(collectEggPrompts()) do
        if isAllowed(e.rarity, e.pet) then
            local part = e.parent
            local pos
            if part and part:IsA("BasePart") then pos = part.Position
            elseif part and part:IsA("Model") then
                local pp = part.PrimaryPart or part:FindFirstChildWhichIsA("BasePart")
                if pp then pos = pp.Position end
            end
            if pos then
                local d = (pos - root.Position).Magnitude
                if d < dist then best, dist = e, d end
            end
        end
    end
    return best
end

local ButterflyCache = {t = 0, list = {}}
local _bfNextScan = 0

local function detectButterflyType(inst)
    if not inst then return nil end
    local text = (inst.Name or ""):lower()
    for _, d in ipairs(inst:GetDescendants()) do
        if d:IsA("StringValue") or d:IsA("TextLabel") then
            text = text .. " " .. tostring(d.Text or d.Value or ""):lower()
        end
    end
    if text:find("radiant") or text:find("gold") then return "Radiant" end
    if text:find("amethyst") or text:find("purple") then return "Amethyst" end
    if text:find("sapphire") or text:find("blue") then return "Sapphire" end
    if text:find("emerald") or text:find("green") then return "Emerald" end
    if text:find("butterfly") then return "Emerald" end
    return nil
end

local function collectButterflies(force)
    local now = tick()
    if not force and not S.AutoButterfly then return ButterflyCache.list end
    if not force and now < _bfNextScan then return ButterflyCache.list end
    local list = {}
    for _, v in ipairs(workspace:GetDescendants()) do
        local n = v.Name:lower()
        if n:find("butterfly") or n:find("bloom") then
            local typ = detectButterflyType(v)
            if typ and S.ButterflyWhitelist[typ] then
                local part = v:IsA("BasePart") and v
                          or (v:IsA("Model") and (v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart")))
                if part then table.insert(list, {inst = v, type = typ, part = part}) end
            end
        end
        if v:IsA("ProximityPrompt") then
            local pn = (v.Name .. " " .. (v.Parent and v.Parent.Name or "")):lower()
            if pn:find("butterfly") or pn:find("catch") or pn:find("net") then
                local typ = detectButterflyType(v.Parent or v)
                if typ and S.ButterflyWhitelist[typ] then
                    local part = v.Parent and (v.Parent:IsA("BasePart") and v.Parent
                               or v.Parent:FindFirstChildWhichIsA("BasePart"))
                    if part then table.insert(list, {inst = v.Parent, type = typ, part = part, prompt = v}) end
                end
            end
        end
    end
    table.sort(list, function(a, b)
        local pa, pb = 99, 99
        for i, t in ipairs(S.ButterflyPriority) do
            if a.type == t then pa = i end
            if b.type == t then pb = i end
        end
        return pa < pb
    end)
    _bfNextScan = now + 1.3 + math.random() * 0.8
    ButterflyCache.t = now
    ButterflyCache.list = list
    return list
end

local function farmTreadmill()
    if not S.AutoTreadmill or S._Stealing then stopMove(); return false end
    local root, hum = hrp(), humanoid()
    if not root or not hum then return false end
    local tm, bestDist = nil, math.huge
    for _, v in ipairs(workspace:GetDescendants()) do
        local n = v.Name:lower()
        if n:find("treadmill", 1, true) or n:find("tread", 1, true) then
            local part = v:IsA("BasePart") and v
                       or (v:IsA("Model") and (v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart")))
            if part then
                local d = (part.Position - root.Position).Magnitude
                if d < bestDist then tm, bestDist = part, d end
            end
        end
    end
    if not tm then return false end
    if bestDist > 9 then
        acWalkSpeed(hum, S.StealWalkSpeed or 16)
        pcall(function() hum:MoveTo(tm.Position) end)
    else
        acWalkSpeed(hum, 16)
        pcall(function() hum:Move(Vector3.new(1, 0, 0), true) end)
    end
    return true
end

local function doStealOnce()
    if S._Stealing then return false end
    S._Stealing = true
    local ok, err = pcall(function()
        local egg = nearestAllowedEgg()
        if not egg then notify("No allowed egg"); return end

        local part = egg.parent
        local cf
        if part:IsA("BasePart") then cf = part.CFrame
        elseif part:IsA("Model") then
            local pp = part.PrimaryPart or part:FindFirstChildWhichIsA("BasePart")
            if pp then cf = pp.CFrame end
        end
        if not cf then return end

        local root = hrp()
        -- approach
        tweenTo(Vector3.new(
            cf.X + jitter(0, -0.9, 0.9),
            root and root.Position.Y or cf.Y,
            cf.Z + jitter(0, -0.9, 0.9)
        ), S.TweenSpeed)
        acWait(0.18)
        firePrompt(egg.prompt)
        notify("Steal → " .. (egg.pet or egg.rarity))

        if S.WebhookOnSteal or (S.WebhookOnRare and
           (egg.rarity == "Secret" or egg.rarity == "Eternal" or egg.rarity == "Divine")) then
            webhook("Egg Stolen", "**Rarity:** " .. egg.rarity .. "\n**Pet:** " .. (egg.pet or "?")
                    .. "\n**Player:** " .. LP.Name, 3066993)
        end
        acWait(jitter(0.22, 0.08, 0.14))

        if S.StealMethod == "HipHeight" then
            -- short air segment to Desert
            tpDesert()
            acWait(jitter(0.22, 0.08, 0.14))
            lowerHip()
            dropEgg()
            acWait(jitter(0.28, 0.10, 0.16))
            local egg2 = nearestAllowedEgg()
            if egg2 and egg2.parent then
                local p2 = egg2.parent
                local cf2 = p2:IsA("BasePart") and p2.CFrame
                           or (p2.PrimaryPart and p2.PrimaryPart.CFrame)
                           or (p2:FindFirstChildWhichIsA("BasePart") and p2:FindFirstChildWhichIsA("BasePart").CFrame)
                if cf2 then
                    local r2 = hrp()
                    tweenTo(Vector3.new(
                        cf2.X + jitter(0, -0.7, 0.7),
                        r2 and r2.Position.Y or cf2.Y,
                        cf2.Z + jitter(0, -0.7, 0.7)
                    ), S.TweenSpeed)
                    acWait(0.15)
                    firePrompt(egg2.prompt)
                    notify("Steal2 → " .. (egg2.pet or egg2.rarity))
                end
            end
            acWait(0.15)
            tpBase()
        else
            -- SoftTween LakeDrop
            tpLake()
            acWait(jitter(0.22, 0.08, 0.14))
            dropEgg()
            acWait(jitter(0.30, 0.10, 0.16))
            local egg2 = nearestAllowedEgg()
            if egg2 and egg2.parent then
                local p2 = egg2.parent
                local cf2 = p2:IsA("BasePart") and p2.CFrame
                           or (p2.PrimaryPart and p2.PrimaryPart.CFrame)
                           or (p2:FindFirstChildWhichIsA("BasePart") and p2:FindFirstChildWhichIsA("BasePart").CFrame)
                if cf2 then
                    local r2 = hrp()
                    tweenTo(Vector3.new(
                        cf2.X + jitter(0, -0.7, 0.7),
                        r2 and r2.Position.Y or cf2.Y,
                        cf2.Z + jitter(0, -0.7, 0.7)
                    ), S.TweenSpeed)
                    acWait(0.15)
                    firePrompt(egg2.prompt)
                    notify("Steal2 → " .. (egg2.pet or egg2.rarity))
                end
            end
        end

        if S.AutoPlace then
            acWait(jitter(0.18, 0.06, 0.12))
            tpBase()
            acWait(0.22)
            tryPlace()
        end
    end)
    S._Stealing = false
    lowerHip()
    if not ok then warn("[A6] steal err:", err) end
    return true
end

local _prioLastRun = {}
local _auxNext = {}

local function _priorityEnabled(label)
    local key = S.FeaturePriorityKey[label]
    return key and S[key] == true
end
local function _priorityIndex(label)
    for i, v in ipairs(S.FeaturePriority) do if v == label then return i end end
    return 99
end
local function _runPriorityFeature(label)
    local key = S.FeaturePriorityKey[label]
    if not key then return end
    _prioLastRun[key] = os.clock()
    if key == "AutoSteal" then if not S._Stealing then pcall(doStealOnce) end
    elseif key == "AutoScramble" then pcall(doScrambleEvent)
    elseif key == "AutoPlace" then pcall(tryPlace)
    elseif key == "AutoHatch" then pcall(tryHatch)
    elseif key == "AutoIndex" then if not S._Stealing then pcall(doStealOnce) end
    elseif key == "AutoTreadmill" then
        if S.AutoTreadmill and not S._Stealing then pcall(farmTreadmill) else stopMove() end
    elseif key == "AutoButterfly" then pcall(catchButterflies)
    end
end

task.spawn(function()
    task.wait(1.6 + math.random() * 1.8)
    while true do
        local anyOn = S.AutoSteal or S.AutoScramble or S.AutoPlace or S.AutoHatch
            or S.AutoIndex or S.AutoTreadmill or S.AutoSell or S.AutoMutation
            or S.AutoUpgradeTreadmill or S.AutoUpgradePlot or S.AutoButterfly
            or S.AutoConvertButterfly or S.AutoEssence
        if not anyOn then task.wait(2.0 + math.random() * 1.8); continue end
        task.wait(0.18 + math.random() * 0.18)
        local now = os.clock()
        local bestLabel, bestScore = nil, -math.huge
        for _, label in ipairs(S.FeaturePriority) do
            local key = S.FeaturePriorityKey[label]
            if key and _priorityEnabled(label) then
                local interval = (S.FeaturePriorityInterval[key] or 1.4) * (0.85 + math.random() * 0.35)
                local age = now - (_prioLastRun[key] or 0)
                if age >= interval then
                    local score = age * 10 - _priorityIndex(label)
                    if score > bestScore then bestScore, bestLabel = score, label end
                end
            end

        if bestLabel then _runPriorityFeature(bestLabel) end

            if not enabled then return end
            local t = _auxNext[key] or 0
            if now >= t then
                _auxNext[key] = now + base * (0.85 + math.random() * 0.50)
                pcall(fn)
            end

        aux("upTM",   S.AutoUpgradeTreadmill, 9.0, upgradeTreadmill)

    end
end)
